import {
  Dialog,
  DialogContent,
  DialogHeader,
  DialogTitle,
  DialogDescription,
  DialogFooter,
} from '@/components/ui/dialog'
import { Button } from '@/components/ui/button'
import { Input } from '@/components/ui/input'
import { Label } from '@/components/ui/label'
import { Switch } from '@/components/ui/switch'
import { Badge } from '@/components/ui/badge'
import { Card, CardContent } from '@/components/ui/card'
import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs'
import { AlertCircle, CheckCircle2, Plus, Trash2, IndianRupee, HelpCircle } from 'lucide-react'
import { toast } from 'sonner'

  code: string
  name: string
  type: 'earning' | 'deduction'
  category: string
  applicable: boolean
  monthlyAmount: number
  annualAmount: number
  calculationMethod: 'FLAT' | 'PERCENTAGE'
  percentageValue?: number
  isStatutory: boolean
  isRecurring: boolean
  effectiveFrom?: string
  effectiveTo?: string
  notes?: string
}

interface CompensationEditorDialogProps {
  open: boolean
  onOpenChange: (open: boolean) => void
  employeeId: string
  employeeCode: string
  employeeName: string
  currentCtc?: number
  currentGross?: number
  existingComponents?: any[]
  onSuccess: () => void
}

export function CompensationEditorDialog({
  open,
  onOpenChange,
  employeeId,
  employeeCode,
  employeeName,
  currentCtc = 0,
  currentGross = 0,
  existingComponents = [],
  onSuccess,
}: CompensationEditorDialogProps) {
  const [annualCtc, setAnnualCtc] = useState<number>(currentCtc)
  const [monthlyGross, setMonthlyGross] = useState<number>(currentGross)
  const [effectiveFrom, setEffectiveFrom] = useState<string>(new Date().toISOString().split('T')[0])
  const [revisionReason, setRevisionReason] = useState<string>('Compensation structure setup')
  const [earnings, setEarnings] = useState<ComponentRow[]>(DEFAULT_EARNINGS)
  const [deductions, setDeductions] = useState<ComponentRow[]>(DEFAULT_DEDUCTIONS)
  const [saving, setSaving] = useState(false)

  // Initialize from props
  useEffect(() => {
    if (open) {
      setAnnualCtc(currentCtc || 0)
      setMonthlyGross(currentGross || 0)
      setEffectiveFrom(new Date().toISOString().split('T')[0])

      // Map existing components if any
      if (existingComponents && existingComponents.length > 0) {
        setEarnings(prev =>
          prev.map(item => {
            const match = existingComponents.find((c: any) => (c.code || c.componentCode)?.toUpperCase() === item.code)
            if (match) {
              const mAmount = Number(match.monthly || match.monthlyAmount || 0)
              return {
                ...item,
                applicable: true,
                monthlyAmount: mAmount,
                annualAmount: Number(match.annual || match.annualAmount || (mAmount * 12)),
              }
            }
            return item
          })
        )

        setDeductions(prev =>
          prev.map(item => {
            const match = existingComponents.find((c: any) => (c.code || c.componentCode)?.toUpperCase() === item.code)
            if (match) {
              const mAmount = Number(match.monthly || match.monthlyAmount || 0)
              return {
                ...item,
                applicable: true,
                monthlyAmount: mAmount,
                annualAmount: Number(match.annual || match.annualAmount || (mAmount * 12)),
              }
            }
            return item
          })
        )
      } else {
        // If no components exist, keep default empty rows
        setEarnings(DEFAULT_EARNINGS.map(e => ({ ...e, monthlyAmount: 0, annualAmount: 0 })))
        setDeductions(DEFAULT_DEDUCTIONS.map(d => ({ ...d, monthlyAmount: 0, annualAmount: 0 })))
      }
    }
  }, [open, currentCtc, currentGross, existingComponents])

  // Calculation summaries
  const totalEarningMonthly = useMemo(() => {
    return earnings
      .filter(e => e.applicable)
      .reduce((sum, e) => sum + (Number(e.monthlyAmount) || 0), 0)
  }, [earnings])

  const totalDeductionsMonthly = useMemo(() => {
    return deductions
      .filter(d => d.applicable)
      .reduce((sum, d) => sum + (Number(d.monthlyAmount) || 0), 0)
  }, [deductions])

  const netPayMonthly = useMemo(() => {
    return Math.max(0, totalEarningMonthly - totalDeductionsMonthly)
  }, [totalEarningMonthly, totalDeductionsMonthly])

  // Gross reconciliation difference
  const grossDifference = useMemo(() => {
    return Math.round((totalEarningMonthly - monthlyGross) * 100) / 100
  }, [totalEarningMonthly, monthlyGross])

  const isReconciled = useMemo(() => {
    return monthlyGross > 0 && Math.abs(grossDifference) <= 0.5
  }, [monthlyGross, grossDifference])

  // Annual CTC reconciliation
  const annualizedGross = useMemo(() => monthlyGross * 12, [monthlyGross])
  const ctcDifference = useMemo(() => annualCtc - annualizedGross, [annualCtc, annualizedGross])

  const handleEarningChange = (index: number, field: keyof ComponentRow, value: any) => {
    setEarnings(prev => {
      const next = [...prev]
      const row = { ...next[index], [field]: value }
      if (field === 'monthlyAmount') {
        row.annualAmount = (Number(value) || 0) * 12
      }
      next[index] = row
      return next
    })
  }

  const handleDeductionChange = (index: number, field: keyof ComponentRow, value: any) => {
    setDeductions(prev => {
      const next = [...prev]
      const row = { ...next[index], [field]: value }
      if (field === 'monthlyAmount') {
        row.annualAmount = (Number(value) || 0) * 12
      }
      next[index] = row
      return next
    })
  }

  const addCustomEarning = () => {
    const code = `CUSTOM_${Date.now()}`
    setEarnings(prev => [
      ...prev,
      {
        code,
        name: 'Custom Allowance',
        type: 'earning',
        category: 'custom',
        applicable: true,
        monthlyAmount: 0,
        annualAmount: 0,
        calculationMethod: 'FLAT',
        isStatutory: false,
        isRecurring: true,
      },
    ])
  }

  const addCustomDeduction = () => {
    const code = `CUSTOM_DED_${Date.now()}`
    setDeductions(prev => [
      ...prev,
      {
        code,
        name: 'Custom Deduction',
        type: 'deduction',
        category: 'custom',
        applicable: true,
        monthlyAmount: 0,
        annualAmount: 0,
        calculationMethod: 'FLAT',
        isStatutory: false,
        isRecurring: true,
      },
    ])
  }

  const handleSave = async (saveAsIncomplete = false) => {
    if (annualCtc <= 0 || monthlyGross <= 0) {
      toast.error('Annual CTC and Monthly Gross must be greater than zero.')
      return
    }

    if (monthlyGross > annualCtc) {
      toast.error('Monthly Gross cannot exceed Annual CTC.')
      return
    }

    if (!saveAsIncomplete && !isReconciled) {
      toast.error(
        `Component total (â‚¹${totalEarningMonthly.toLocaleString('en-IN')}) does not match Monthly Gross (â‚¹${monthlyGross.toLocaleString('en-IN')}). Reconcile component sum before saving.`
      )
      return
    }

    setSaving(true)
    try {
      const componentsToSave: any[] = []

      if (!saveAsIncomplete) {
        // Collect active earnings
        earnings.filter(e => e.applicable && e.monthlyAmount > 0).forEach(e => {
          componentsToSave.push({
            code: e.code,
            name: e.name,
            type: 'earning',
            category: e.category,
            monthly_amount: Number(e.monthlyAmount),
            annual_amount: Number(e.annualAmount || e.monthlyAmount * 12),
            is_active: true,
            is_statutory: false,
            notes: e.notes || null,
          })
        })

        // Collect active deductions
        deductions.filter(d => d.applicable && d.monthlyAmount > 0).forEach(d => {
          componentsToSave.push({
            code: d.code,
            name: d.name,
            type: 'deduction',
            category: d.category,
            monthly_amount: Number(d.monthlyAmount),
            annual_amount: Number(d.annualAmount || d.monthlyAmount * 12),
            is_active: true,
            is_statutory: d.isStatutory,
            notes: d.notes || null,
          })
        })
      }

      const res = await fetch(`/api/compensation/${employeeId}`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          annual_ctc: annualCtc,
          monthly_gross: monthlyGross,
          effective_from: effectiveFrom,
          reason: revisionReason,
          components: componentsToSave,
        }),
      })

      const data = await res.json()
      if (data.success) {
        toast.success(
          saveAsIncomplete
            ? 'Compensation saved as Incomplete (Payroll remains blocked until components configured).'
            : 'Compensation structure saved and reconciled successfully!'
        )
        onOpenChange(false)
        onSuccess()
      } else {
        toast.error(data.error || 'Failed to save compensation structure.')
      }
    } catch (err: any) {
      toast.error(err.message || 'Error saving compensation structure.')
    } finally {
      setSaving(false)
    }
  }

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent className="max-w-4xl max-h-[90vh] flex flex-col p-6">
        <DialogHeader className="pb-3 border-b">
          <div className="flex items-center justify-between">
            <div>
              <DialogTitle className="text-base font-bold">Edit Compensation Structure</DialogTitle>
              <DialogDescription className="text-xs">
                {employeeCode} â€” {employeeName} | Enterprise Canonical Salary Configuration
              </DialogDescription>
            </div>
            {isReconciled ? (
              <Badge className="bg-emerald-500/15 text-emerald-600 dark:text-emerald-400 border-emerald-500/30 gap-1 text-[11px]">
                <CheckCircle2 className="w-3.5 h-3.5" /> Reconciled
              </Badge>
            ) : (
              <Badge variant="destructive" className="gap-1 text-[11px]">
                <AlertCircle className="w-3.5 h-3.5" /> Reconciliation Required
              </Badge>
            )}
          </div>
        </DialogHeader>

        <div className="flex-1 overflow-y-auto space-y-4 py-3 text-xs pr-1">
          {/* Top Level Structure Inputs */}
          <div className="grid grid-cols-1 sm:grid-cols-4 gap-3 p-3 rounded-lg bg-muted/40 border">
            <div>
              <Label className="text-[11px] text-muted-foreground">Annual CTC (â‚¹)</Label>
              <Input
                type="number"
                value={annualCtc || ''}
                onChange={e => setAnnualCtc(Number(e.target.value) || 0)}
                placeholder="691200"
                className="h-8 text-xs font-semibold mt-1"
              />
            </div>
            <div>
              <Label className="text-[11px] text-muted-foreground">Monthly Gross (â‚¹)</Label>
              <Input
                type="number"
                value={monthlyGross || ''}
                onChange={e => setMonthlyGross(Number(e.target.value) || 0)}
                placeholder="57600"
                className="h-8 text-xs font-semibold mt-1"
              />
            </div>
            <div>
              <Label className="text-[11px] text-muted-foreground">Effective From</Label>
              <Input
                type="date"
                value={effectiveFrom}
                onChange={e => setEffectiveFrom(e.target.value)}
                className="h-8 text-xs mt-1"
              />
            </div>
            <div>
              <Label className="text-[11px] text-muted-foreground">Revision Reason</Label>
              <Input
                type="text"
                value={revisionReason}
                onChange={e => setRevisionReason(e.target.value)}
                placeholder="e.g. Annual Appraisal / Correction"
                className="h-8 text-xs mt-1"
              />
            </div>
          </div>

          {/* Gross Reconciliation Alert Banner */}
          {!isReconciled && (
            <div className="p-3 rounded-lg bg-destructive/10 border border-destructive/30 flex items-start gap-2.5">
              <AlertCircle className="w-4 h-4 text-destructive shrink-0 mt-0.5" />
              <div className="text-xs space-y-1">
                <p className="font-semibold text-destructive">
                  Component total does not match Monthly Gross
                </p>
                <p className="text-muted-foreground">
                  Active earning components sum to{' '}
                  <span className="font-semibold text-foreground">
                    â‚¹{totalEarningMonthly.toLocaleString('en-IN')}
                  </span>
                  , but Monthly Gross target is{' '}
                  <span className="font-semibold text-foreground">
                    â‚¹{monthlyGross.toLocaleString('en-IN')}
                  </span>
                  . Difference:{' '}
                  <span className="font-bold text-destructive">
                    â‚¹{Math.abs(grossDifference).toLocaleString('en-IN')} {grossDifference > 0 ? '(excess)' : '(short)'}
                  </span>
                  . Adjust components before saving.
                </p>
              </div>
            </div>
          )}

          {/* Components Tab: Earnings vs Deductions */}
          <Tabs defaultValue="earnings" className="w-full">
            <TabsList className="grid grid-cols-2 w-full h-8">
              <TabsTrigger value="earnings" className="text-xs">
                Earnings ({earnings.filter(e => e.applicable).length}) â€” Total: â‚¹{totalEarningMonthly.toLocaleString('en-IN')}
              </TabsTrigger>
              <TabsTrigger value="deductions" className="text-xs">
                Deductions ({deductions.filter(d => d.applicable).length}) â€” Total: â‚¹{totalDeductionsMonthly.toLocaleString('en-IN')}
              </TabsTrigger>
            </TabsList>

            {/* TAB 1: EARNINGS */}
            <TabsContent value="earnings" className="mt-3 space-y-2">
              <div className="flex items-center justify-between pb-1">
                <span className="text-[11px] text-muted-foreground">
                  Configure monthly earning components. Sum must reconcile exactly to â‚¹{monthlyGross.toLocaleString('en-IN')}.
                </span>
                <Button variant="outline" size="sm" onClick={addCustomEarning} className="h-7 text-xs gap-1">
                  <Plus className="w-3 h-3" /> Add Custom Earning
                </Button>
              </div>

              <div className="border rounded-lg overflow-hidden divide-y">
                <div className="grid grid-cols-12 gap-2 p-2 bg-muted/50 font-semibold text-[11px] text-muted-foreground">
                  <div className="col-span-1 text-center">Active</div>
                  <div className="col-span-4">Component Name</div>
                  <div className="col-span-3 text-right">Monthly (â‚¹)</div>
                  <div className="col-span-3 text-right">Annual (â‚¹)</div>
                  <div className="col-span-1 text-center">Notes</div>
                </div>

                {earnings.map((row, idx) => (
                  <div
                    key={row.code + idx}
                    className={`grid grid-cols-12 gap-2 p-2 items-center transition-colors ${
                      row.applicable ? 'bg-background' : 'bg-muted/20 opacity-60'
                    }`}
                  >
                    <div className="col-span-1 flex justify-center">
                      <Switch
                        checked={row.applicable}
                        onCheckedChange={v => handleEarningChange(idx, 'applicable', v)}
                      />
                    </div>
                    <div className="col-span-4">
                      {row.code.startsWith('CUSTOM_') ? (
                        <Input
                          type="text"
                          value={row.name}
                          onChange={e => handleEarningChange(idx, 'name', e.target.value)}
                          className="h-7 text-xs"
                        />
                      ) : (
                        <div>
                          <span className="font-medium block text-foreground">{row.name}</span>
                          <span className="text-[10px] text-muted-foreground font-mono">{row.code}</span>
                        </div>
                      )}
                    </div>
                    <div className="col-span-3">
                      <Input
                        type="number"
                        disabled={!row.applicable}
                        value={row.monthlyAmount || ''}
                        onChange={e => handleEarningChange(idx, 'monthlyAmount', Number(e.target.value) || 0)}
                        className="h-7 text-xs text-right font-medium"
                        placeholder="0"
                      />
                    </div>
                    <div className="col-span-3 text-right pr-2 font-mono text-[11px]">
                      â‚¹{(Number(row.annualAmount) || 0).toLocaleString('en-IN')}
                    </div>
                    <div className="col-span-1 flex justify-center">
                      {row.code.startsWith('CUSTOM_') ? (
                        <Button
                          variant="ghost"
                          size="icon"
                          className="h-6 w-6 text-destructive"
                          onClick={() => setEarnings(prev => prev.filter((_, i) => i !== idx))}
                        >
                          <Trash2 className="w-3.5 h-3.5" />
                        </Button>
                      ) : (
                        <span className="text-[10px] text-muted-foreground">{row.category}</span>
                      )}
                    </div>
                  </div>
                ))}
              </div>
            </TabsContent>

            {/* TAB 2: DEDUCTIONS */}
            <TabsContent value="deductions" className="mt-3 space-y-2">
              <div className="flex items-center justify-between pb-1">
                <span className="text-[11px] text-muted-foreground">
                  Configure employee-specific deductions. If PT is not configured, leave amount as 0 to flag as NOT CONFIGURED.
                </span>
                <Button variant="outline" size="sm" onClick={addCustomDeduction} className="h-7 text-xs gap-1">
                  <Plus className="w-3 h-3" /> Add Custom Deduction
                </Button>
              </div>

              <div className="border rounded-lg overflow-hidden divide-y">
                <div className="grid grid-cols-12 gap-2 p-2 bg-muted/50 font-semibold text-[11px] text-muted-foreground">
                  <div className="col-span-1 text-center">Active</div>
                  <div className="col-span-4">Deduction Name</div>
                  <div className="col-span-3 text-right">Monthly (â‚¹)</div>
                  <div className="col-span-3 text-right">Annual (â‚¹)</div>
                  <div className="col-span-1 text-center">Type</div>
                </div>

                {deductions.map((row, idx) => (
                  <div
                    key={row.code + idx}
                    className={`grid grid-cols-12 gap-2 p-2 items-center transition-colors ${
                      row.applicable ? 'bg-background' : 'bg-muted/20 opacity-60'
                    }`}
                  >
                    <div className="col-span-1 flex justify-center">
                      <Switch
                        checked={row.applicable}
                        onCheckedChange={v => handleDeductionChange(idx, 'applicable', v)}
                      />
                    </div>
                    <div className="col-span-4">
                      {row.code.startsWith('CUSTOM_DED_') ? (
                        <Input
                          type="text"
                          value={row.name}
                          onChange={e => handleDeductionChange(idx, 'name', e.target.value)}
                          className="h-7 text-xs"
                        />
                      ) : (
                        <div>
                          <span className="font-medium block text-foreground">{row.name}</span>
                          <span className="text-[10px] text-muted-foreground">
                            {row.notes || row.code}
                          </span>
                        </div>
                      )}
                    </div>
                    <div className="col-span-3">
                      <Input
                        type="number"
                        disabled={!row.applicable}
                        value={row.monthlyAmount || ''}
                        onChange={e => handleDeductionChange(idx, 'monthlyAmount', Number(e.target.value) || 0)}
                        className="h-7 text-xs text-right font-medium text-destructive"
                        placeholder="0"
                      />
                    </div>
                    <div className="col-span-3 text-right pr-2 font-mono text-[11px] text-destructive">
                      â‚¹{(Number(row.annualAmount) || 0).toLocaleString('en-IN')}
                    </div>
                    <div className="col-span-1 flex justify-center">
                      {row.code.startsWith('CUSTOM_DED_') ? (
                        <Button
                          variant="ghost"
                          size="icon"
                          className="h-6 w-6 text-destructive"
                          onClick={() => setDeductions(prev => prev.filter((_, i) => i !== idx))}
                        >
                          <Trash2 className="w-3.5 h-3.5" />
                        </Button>
                      ) : (
                        <span className="text-[10px] text-muted-foreground">
                          {row.isStatutory ? 'Statutory' : 'Custom'}
                        </span>
                      )}
                    </div>
                  </div>
                ))}
              </div>
            </TabsContent>
          </Tabs>

          {/* Reconciliation & Net Summary Card */}
          <div className="grid grid-cols-1 sm:grid-cols-3 gap-3 p-3 rounded-lg bg-muted/40 border">
            <div>
              <span className="text-muted-foreground block text-[10px]">Monthly Gross Earning</span>
              <span className="text-sm font-bold text-foreground">
                â‚¹{totalEarningMonthly.toLocaleString('en-IN')}
              </span>
              <span className="text-[10px] text-muted-foreground block mt-0.5">
                Target: â‚¹{monthlyGross.toLocaleString('en-IN')}
              </span>
            </div>
            <div>
